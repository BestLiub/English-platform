# English 学习站改造 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 v0.1 扩成三个模块的学习站：主页三入口、背单词新增沉浸模式、计划模式补复习上限、统计与设置合并成个人页。

**Architecture:** 沉浸模式独立成 `immersion_word` 表 + 独立 service + 独立路由，与现有的 `user_word` / FSRS 完全隔离。前端新增一个沉浸模式页面和共用的模式切换条，计划模式的代码基本不动，只增加复习上限参数。

**Tech Stack:** Vue 3 + TypeScript + Pinia + Tailwind（前端）、FastAPI + SQLAlchemy 2.0 + Alembic + SQLite（后端）、pytest（后端测试，本轮新增）

**Spec:** `docs/superpowers/specs/2026-11-03-english-platform-design.md`

## Global Constraints

- 单用户：所有表的 `user_id` 固定为 `1`，但字段必须存在
- 数据库：SQLite，模式变更一律走 Alembic 迁移，不要手改表结构
- 界面文案用中文，风格沿用现有的浅灰 / 深灰单色系，不引入彩色主色
- 后端不新增重依赖；测试框架只加 `pytest`
- 前端没有单元测试框架，验证手段是 `npm run typecheck` + 浏览器实操
- 沉浸模式和计划模式的数据不得互相写入
- **仓库目前还不是 git 仓库**。在用户确认 `git init` 之前，所有「提交」步骤跳过，改为在任务结束时汇报改了哪些文件

## Review Focus

这些是规格里没明说、但真用起来最容易出问题的输入，各任务里要有对应的测试：

1. **词库为空或全部掌握时抽词** —— 不能 500，要返回 `finished: true`，界面显示完成页
2. **剩余池小于防重复窗口** —— 不能抽不出词，必须有回退
3. **一个词同时属于多个词库**（如 cet4 + cet6）—— 在六级跳过它，重置六级不能把四级的进度也清掉，但它在六级确实不该再出现
4. **复习上限设成 0** —— 应该只出复习 0 张，新词照常，而不是报错或全量
5. **沉浸模式跳过的词出现在计划模式新词里** —— 必须是允许的（两套数据隔离）

```python
    last_seen_at: Mapped[datetime] = mapped_column(default=utcnow, index=True)

    __table_args__ = (UniqueConstraint("user_id", "word_id", name="uq_immersion_word"),)
```

`api/app/services/immersion.py`：

```python
"""沉浸模式：随机抽词，与 FSRS 完全隔离。"""
from __future__ import annotations

from sqlalchemy import func, select
from sqlalchemy.orm import Session

from ..models import ImmersionWord, Word
from .srs import utcnow

MAX_REPEAT_WINDOW = 50


def _book_filter(wordbook: str):
    return Word.tag.like(f"%{wordbook}%")


def _mastered_ids(user_id: int):
    return select(ImmersionWord.word_id).where(
        ImmersionWord.user_id == user_id,
        ImmersionWord.status == "mastered",
    )


def candidate_count(db: Session, wordbook: str, user_id: int = 1) -> int:
    return int(
        db.scalar(
            select(func.count())
            .select_from(Word)
            .where(_book_filter(wordbook), Word.id.not_in(_mastered_ids(user_id)))
        )
        or 0
    )


def repeat_window(count: int) -> int:
    """最近这么多个词不进候选，池子小时按比例缩小，永不等于池子大小。"""
    return min(MAX_REPEAT_WINDOW, count // 3)


def draw(db: Session, wordbook: str, user_id: int = 1) -> Word | None:
    remaining = candidate_count(db, wordbook, user_id)
    if remaining == 0:
        return None

    base = select(Word).where(_book_filter(wordbook), Word.id.not_in(_mastered_ids(user_id)))

    window = repeat_window(remaining)
    word = None
    if window:
        recent = (
            select(ImmersionWord.word_id)
            .where(ImmersionWord.user_id == user_id)
            .order_by(ImmersionWord.last_seen_at.desc())
            .limit(window)
        )
        word = db.scalar(
            base.where(Word.id.not_in(recent)).order_by(func.random()).limit(1)
        )
    if word is None:                      # 窗口把候选全挡住了，退回去不过滤
        word = db.scalar(base.order_by(func.random()).limit(1))
    if word is None:
        return None

    _touch(db, word.id, user_id)
    return word


def _touch(db: Session, word_id: int, user_id: int = 1) -> None:
    row = db.scalar(
        select(ImmersionWord).where(
            ImmersionWord.user_id == user_id, ImmersionWord.word_id == word_id
        )
    )
    if row is None:
        db.add(ImmersionWord(user_id=user_id, word_id=word_id, status="seen", last_seen_at=utcnow()))
    else:
        row.last_seen_at = utcnow()
    db.commit()
```

同时把 `ImmersionWord` 从 `app.models` 导出，测试里 `immersion.ImmersionWord` 需要能取到——在 service 顶部 `from ..models import ImmersionWord` 已经在命名空间里。

```python
def mark_mastered(db: Session, word_id: int, user_id: int = 1) -> None:
    row = db.scalar(
        select(ImmersionWord).where(
            ImmersionWord.user_id == user_id, ImmersionWord.word_id == word_id
        )
    )
    if row is None:
        db.add(ImmersionWord(user_id=user_id, word_id=word_id, status="mastered"))
    else:
        row.status = "mastered"
    db.commit()


def progress(db: Session, wordbook: str, user_id: int = 1) -> dict:
    total = int(
        db.scalar(select(func.count()).select_from(Word).where(_book_filter(wordbook))) or 0
    )
    mastered = int(
        db.scalar(
            select(func.count())
            .select_from(ImmersionWord)
            .join(Word, Word.id == ImmersionWord.word_id)
            .where(
                ImmersionWord.user_id == user_id,
                ImmersionWord.status == "mastered",
                _book_filter(wordbook),
            )
        )
        or 0
    )
    return {"total": total, "mastered": mastered, "remaining": total - mastered}


def reset(db: Session, wordbook: str, user_id: int = 1) -> int:
    book_ids = select(Word.id).where(_book_filter(wordbook))
    result = db.execute(
        delete(ImmersionWord).where(
            ImmersionWord.user_id == user_id, ImmersionWord.word_id.in_(book_ids)
        )
    )
    db.commit()
    return int(result.rowcount or 0)
```

顶部补 `from sqlalchemy import delete, func, select`。

```python
from __future__ import annotations

from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.orm import Session

from ..db import get_db
from ..models import Word
from ..schemas.immersion import NextRequest, ResetRequest, SkipRequest
from ..services import immersion

router = APIRouter(prefix="/immersion", tags=["immersion"])

USER_ID = 1


def _item(word: Word) -> dict:
    return {
        "word_id": word.id,
        "word": word.word,
        "phonetic": word.phonetic,
        "translation": word.translation,
    }


@router.post("/next")
def next_word(payload: NextRequest, db: Session = Depends(get_db)) -> dict:
    word = immersion.draw(db, payload.wordbook, USER_ID)
    state = immersion.progress(db, payload.wordbook, USER_ID)
    return {
        "code": 0,
        "message": "ok",
        "data": {
            "word": _item(word) if word else None,
            "progress": state,
            "finished": word is None,
        },
    }


@router.post("/skip")
def skip(payload: SkipRequest, db: Session = Depends(get_db)) -> dict:
    if db.get(Word, payload.word_id) is None:
        raise HTTPException(status_code=404, detail="词条不存在")
    immersion.mark_mastered(db, payload.word_id, USER_ID)
    return {"code": 0, "message": "ok", "data": immersion.progress(db, payload.wordbook, USER_ID)}


@router.post("/reset")
def reset(payload: ResetRequest, db: Session = Depends(get_db)) -> dict:
    immersion.reset(db, payload.wordbook, USER_ID)
    return {"code": 0, "message": "ok", "data": immersion.progress(db, payload.wordbook, USER_ID)}


@router.get("/stats")
def stats(wordbook: str = Query("cet4"), db: Session = Depends(get_db)) -> dict:
    return {"code": 0, "message": "ok", "data": immersion.progress(db, wordbook, USER_ID)}
```

- [ ] **Step 2: 写 store**

`web/src/stores/immersion.ts`：

```ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import { fetchNext, resetImmersion, skipWord, type ImmersionProgress, type ImmersionWord } from '@/api/immersion'

export const useImmersionStore = defineStore('immersion', () => {
  const word = ref<ImmersionWord | null>(null)
  const progress = ref<ImmersionProgress>({ total: 0, mastered: 0, remaining: 0 })
  const loading = ref(false)
  const submitting = ref(false)
  const finished = ref(false)

  async function next(wordbook: string) {
    loading.value = true
    try {
      const result = await fetchNext(wordbook)
      word.value = result.word
      progress.value = result.progress
      finished.value = result.finished
    } finally {
      loading.value = false
    }
  }

  async function skip(wordbook: string) {
    if (!word.value || submitting.value) return
    submitting.value = true
    try {
      progress.value = await skipWord(wordbook, word.value.word_id)
      await next(wordbook)
    } finally {
      submitting.value = false
    }
  }

  async function restart(wordbook: string) {
    submitting.value = true
    try {
      progress.value = await resetImmersion(wordbook)
      finished.value = false
      await next(wordbook)
    } finally {
      submitting.value = false
    }
  }

  return { word, progress, loading, submitting, finished, next, skip, restart }
})
```
